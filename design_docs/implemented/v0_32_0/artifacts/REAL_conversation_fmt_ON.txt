module benchmark/solution

import std/io (println)

pure func contains(x: int, xs: list[int]) -> bool = match xs {
  [] => false,
   ::(h, t) => if x == h then true else contains(x, t)
}

pure func getNeighbors(node: int, graph: list[(int, list[int])]) -> list[int] = match graph {
  [] => [],
   ::((n, neighbors), rest) => if n == node then neighbors else getNeighbors(node, rest)
}

func bfs_visit(queue: list[int], visited: list[int], result: list[int], graph: list[(int, list[int])]) -> list[int] = match queue {
  [] => result,
   ::(node, rest) => if contains(node, visited) then bfs_visit(rest, visited, result, graph) else {

}
